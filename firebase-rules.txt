/* ============================================================================
   우리팀 허브 — Firebase 보안 규칙 (Firestore Rules)
   ----------------------------------------------------------------------------
   ▶ 적용 위치
     Firebase 콘솔 → Build → Firestore Database → 규칙(Rules) 탭에 이 파일 전체를
     붙여넣고 [게시] 합니다.

   ▶ 최초 1회 설정 (무료 Spark 요금제 기준, 카드 등록 불필요)
     1) Authentication → Sign in method → "이메일/비밀번호" 사용 설정
        (앱은 사번을 사번@teamhub.local 로 바꿔서 이 방식을 그대로 씀니다)
     2) Authentication → Settings → 승인된 도메인에 앱 주소 추가
        (예: 내아이디.github.io)
     3) Firestore Database 만들기 (위치: asia-northeast3 서울 권장)
     4) 첫 관리자 지정 — 둘 중 하나
        (a) 앱에서 가장 먼저 회원가입(이름·사번·직책·비밀번호)한 뒤, 콘솔 users/{uid} 문서에서
            role = "admin", status = "approved" 로 수정
        (b) 아래 BOOTSTRAP 에 그 사람의 가상 이메일(사번@teamhub.local)을 넣으면
            그 사번으로 가입할 때 처음부터 admin / approved 로 만들어집니다.
     5) 이 파일을 게시

   ▶ 데이터 구조
     users/{uid}            { empId, email(=사번@teamhub.local), name, position,
                              role:'admin'|'member', canManageDuty,
                              status:'pending'|'approved'|'rejected',
                              workStatus, statusNote, statusUpdatedAt, createdAt }
                            → 로그인은 사번 + 비밀번호. 사번은 내부에서만
                              사번@teamhub.local 가상 이메일로 변환해 Auth 에 넘깁니다.
                            → position(직책: 팀장·주무관·주임…)과 앱 권한은 별개
                              role = 'admin'        : 관리자(승인·권한 부여)
                              canManageDuty = true  : 당직 관리(명단·재배정·행사일·자동 배정)
     announcements/{id}     { title, body, is_pinned, is_important, author_id,
                              author_name, createdAt }
       └ comments/{cid}     { author_id, author_name, body, createdAt }
       └ reads/{uid}        { uid, name, at }
     duty_schedules/{id}    { startDate:'YYYY-MM-DD', endDate:'YYYY-MM-DD',
                              type:'day_island'|'overnight_island'|'duty'|'edu'
                                   |'board_meeting'|'opinion_due'|'survey_due',
                              memberId, memberName, memo, createdBy, createdAt,
                              history:[{fromName,toName,editedByName,editedAt}],
                              previousMemberName, updatedByName, updatedAt }
                              → 당직 담당자를 바꾼 이력(누가·언제) 누적
                              → board_meeting(등판) · opinion_due(의소기한) · survey_due(조사기한)은
                                당직 달력 아래 영역. 규칙 밖 날짜에 직접 넣은 것만 여기 저장되고
                                (id = 날짜_ms_종류, memberId = 'ms'), 규칙 날짜는 앱이 자동 계산
     duty_roster_config/config
                            { excluded:[uid,...], customMembers:[{id,name},...],
                              msOff:['YYYY-MM-DD|종류',...], teamName:'우리팀 이름',
                              updatedBy, updatedAt }
                              → 당직 명단(가나다순): 제외한 uid + 회원이 아닌 직접 추가 이름
                              → msOff: 자동 등판·의소기한·조사기한 중 뺀 날짜(다시 뜨지 않게)
                              → teamName: 관리자가 [팀원관리 → 팀 이름]에서 바꾸면 모든 팀원 화면에
                                실시간 반영. 필드가 하나 늘었을 뿐 규칙 내용은 그대로라
                                이 파일을 다시 게시하지 않아도 됩니다.
     team_events/{date}     { date:'YYYY-MM-DD', title, createdBy, createdAt }
                              → 행사일(당직 없음). 문서 id = 날짜
     photos/{id}            { imageData(Data URL), caption, author_id, author_name,
                              likes:[uid], comments:[{...}], createdAt }
     chat_messages/{id}     { author_id, author_name, body, createdAt }

   ▶ 원칙
     - 승인(status == 'approved')된 팀원만 공지·일정·사진·대화를 읽고 쓸 수 있습니다.
     - 승인 대기(pending)·거절(rejected) 상태는 자기 프로필만 볼 수 있습니다.
     - 직책(position)은 표시용이고, 실제 권한은 role 과 canManageDuty 로만 정해집니다.
     - 권한(role·canManageDuty)·승인 상태(status) 변경은 관리자(role == 'admin')만 가능합니다.
     - 일정은 본인 것만 등록/수정/삭제할 수 있고, 관리자는 전원 일정을 관리합니다.
     - 당직(type == 'duty') 일정은 팀원끼리 상의해 서로 바꿀 수 있도록 승인 팀원 누구나
       담당자를 만들고 바꿀 수 있습니다. 바꿀 때마다 history 에 누가·언제·누구를 누구로
       바꿨는지가 쌓입니다.
     - 당직 명단(duty_roster_config)·행사일(team_events)은 관리자와 당직 관리 권한자가 관리합니다.
     - 등판(board_meeting)·의소기한(opinion_due)·조사기한(survey_due)은 팀 전체 일정이라
       관리자와 당직 관리 권한자가 등록·삭제합니다(규칙 날짜는 앱이 자동 계산).
     - 사진은 Storage 없이 Firestore 문서(imageData)로 저장 → Spark 요금제 호환.
   ============================================================================ */

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    /* ---------- 부트스트랩 관리자 (가상 이메일) ----------
       최초 관리자로 만들 사번을 사번@teamhub.local 형태로 적어두세요. (없으면 빈 문자열 유지)
       예: 'kim@example.com'
       현재 지정: 사번 250620 → '250620@teamhub.local'            */
    function isBootstrap() {
      return signedIn() && request.auth.token.email == '250620@teamhub.local';
    }

    /* ---------- 공통 헬퍼 ---------- */
    function signedIn() {
      return request.auth != null;
    }
    function userPath() {
      return /databases/$(database)/documents/users/$(request.auth.uid);
    }
    function myData() {
      return get(userPath()).data;
    }
    function myProfileExists() {
      return signedIn() && exists(userPath());
    }
    /* 승인된 팀원인가 — 본인 프로필 문서를 읽어 판단 */
    function isApproved() {
      return myProfileExists() && myData().status == 'approved';
    }
    function isAdmin() {
      return myProfileExists() && myData().role == 'admin' && myData().status == 'approved';
    }
    /* 당직 관리 권한 — 관리자이거나 관리자가 canManageDuty 를 부여한 팀원 */
    function canManageDuty() {
      return isApproved()
        && (myData().role == 'admin'
            || (('canManageDuty' in myData()) && myData().canManageDuty == true));
    }
    function isSelf(uid) {
      return signedIn() && request.auth.uid == uid;
    }
    /* 변경된 필드가 지정한 목록 안에만 있는가 */
    function onlyChanged(keys) {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys);
    }
    function validStatus(s) {
      return s in ['', 'day_island', 'overnight_island', 'duty', 'edu'];
    }
    function validEntryType(t) {
      return t in ['day_island', 'overnight_island', 'duty', 'edu',
                   'board_meeting', 'opinion_due', 'survey_due'];
    }
    /* 등판·의소기한·조사기한 — 팀 전체 일정(당직 달력 아래 영역) */
    function isMilestoneType(t) {
      return t in ['board_meeting', 'opinion_due', 'survey_due'];
    }
    /* 담당자 변경 이력 필드 검증 */
    function validHistory(d) {
      return (!('history' in d) || (d.history is list && d.history.size() <= 30))
        && (!('previousMemberName' in d) || d.previousMemberName is string)
        && (!('updatedByName' in d) || d.updatedByName is string)
        && (!('updatedAt' in d) || d.updatedAt is int);
    }

    /* =====================================================================
       users — 프로필 · 승인 상태 · 근무 상태
       ===================================================================== */
    match /users/{uid} {
      /* 승인된 팀원은 팀원 목록을 볼 수 있고, 본인 문서는 항상 볼 수 있음 */
      allow read: if isSelf(uid) || isApproved();

      /* 가입 직후 본인 문서 생성 — 항상 pending/member 로만 생성 가능
         (BOOTSTRAP 이메일은 예외적으로 admin/approved 생성 허용) */
      allow create: if isSelf(uid)
        && request.resource.data.keys().hasOnly(
             ['empId','email','name','position','role','status','canManageDuty','workStatus','statusNote','statusUpdatedAt','createdAt'])
        && (
             (request.resource.data.status == 'pending' && request.resource.data.role == 'member'
                && (!('canManageDuty' in request.resource.data) || request.resource.data.canManageDuty == false))
             || (isBootstrap() && request.resource.data.status == 'approved' && request.resource.data.role == 'admin')
           )
        && request.resource.data.empId is string
        && request.resource.data.empId.matches('^[0-9A-Za-z_-]{2,20}$')
        && request.resource.data.position is string
        && (request.resource.data.position == ''
            || request.resource.data.position in ['주임','대리','과장','팀장','센터장','부장'])
        && request.resource.data.email is string
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 30;

      /* 관리자: 누구든 role/status/canManageDuty 변경 가능 (권한 부여·해제)
         본인: 이름·직책·근무상태만 변경 가능 (사번/role/status/canManageDuty 는 절대 못 바꿈) */
      allow update: if isAdmin()
        || (isSelf(uid)
            && onlyChanged(['name','position','workStatus','statusNote','statusUpdatedAt'])
            && request.resource.data.role == resource.data.role
            && request.resource.data.status == resource.data.status
            && (!('empId' in request.resource.data) || !('empId' in resource.data)
                || request.resource.data.empId == resource.data.empId)
            && (!('canManageDuty' in request.resource.data)
                || request.resource.data.canManageDuty == (('canManageDuty' in resource.data) && resource.data.canManageDuty == true))
            && validStatus(request.resource.data.workStatus));

      /* 관리자만 팀원 삭제(내보내기) — 본인 삭제는 불가 */
      allow delete: if isAdmin() && !isSelf(uid);
    }

    /* =====================================================================
       announcements — 공지사항
       ===================================================================== */
    match /announcements/{id} {
      allow read: if isApproved();

      allow create: if isApproved()
        && request.resource.data.author_id == request.auth.uid
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.title.size() <= 60
        && request.resource.data.body is string
        && request.resource.data.body.size() <= 20000;

      allow update: if isApproved()
        && (resource.data.author_id == request.auth.uid || isAdmin())
        && onlyChanged(['title','body','is_pinned','is_important','updatedAt','author_id','author_name']);

      allow delete: if isApproved()
        && (resource.data.author_id == request.auth.uid || isAdmin());

      /* 댓글 */
      match /comments/{cid} {
        allow read: if isApproved();
        allow create: if isApproved()
          && request.resource.data.author_id == request.auth.uid
          && request.resource.data.body is string
          && request.resource.data.body.size() > 0
          && request.resource.data.body.size() <= 1000;
        allow delete: if isApproved()
          && (resource.data.author_id == request.auth.uid || isAdmin());
        allow update: if false;
      }

      /* 읽음 확인 */
      match /reads/{readerId} {
        allow read: if isApproved();
        allow create, update: if isApproved() && readerId == request.auth.uid;
        allow delete: if false;
      }
    }

    /* =====================================================================
       duty_schedules — 팀 캘린더(당직 · 섬 출장 · 교육) 일정
       ===================================================================== */
    match /duty_schedules/{id} {
      allow read: if isApproved();

      /* 본인 일정만 등록. 관리자는 전원, 당직(type 'duty')은 승인 팀원 누구나 지정 가능(팀원끼리 교대) */
      allow create: if isApproved()
        && (request.resource.data.memberId == request.auth.uid
            || isAdmin()
            || request.resource.data.type == 'duty'
            || (isMilestoneType(request.resource.data.type) && canManageDuty()))
        && validEntryType(request.resource.data.type)
        && validHistory(request.resource.data)
        && request.resource.data.startDate is string
        && request.resource.data.endDate is string
        && request.resource.data.startDate.size() == 10
        && request.resource.data.endDate >= request.resource.data.startDate
        && request.resource.data.memberName is string
        && request.resource.data.memo is string
        && request.resource.data.memo.size() <= 200;

      /* 본인 것만 수정 (기존 소유자 기준), 관리자는 전원, 당직은 승인 팀원 누구나 */
      allow update: if isApproved()
        && (resource.data.memberId == request.auth.uid
            || isAdmin()
            || (resource.data.type == 'duty' && request.resource.data.type == 'duty')
            || (resource.data.type == request.resource.data.type
                && isMilestoneType(resource.data.type) && canManageDuty()))
        && (request.resource.data.memberId == request.auth.uid || isAdmin()
            || request.resource.data.type == 'duty'
            || (isMilestoneType(request.resource.data.type) && canManageDuty()))
        && validEntryType(request.resource.data.type)
        && validHistory(request.resource.data)
        && request.resource.data.endDate >= request.resource.data.startDate;

      allow delete: if isApproved()
        && (resource.data.memberId == request.auth.uid
            || isAdmin()
            || resource.data.type == 'duty'
            || (isMilestoneType(resource.data.type) && canManageDuty()));
    }

    /* =====================================================================
       duty_roster_config — 당직 명단 설정
       · 승인된 팀원이 가나다순으로 순번을 이루고, excluded 의 uid 는 명단에서 뺍니다.
       · customMembers 는 회원 가입을 하지 않은 사람도 당직 순번에 넣기 위한 직접 추가 이름입니다.
       · 명단 변경은 관리자 또는 당직 관리 권한자만 (달력의 [🛡️ 명단] 시트)
       ===================================================================== */
    match /duty_roster_config/{docId} {
      allow read: if isApproved();
      allow create, update: if canManageDuty()
        && request.resource.data.excluded is list
        && request.resource.data.excluded.size() <= 50
        && (!('customMembers' in request.resource.data) || (
              request.resource.data.customMembers is list
              && request.resource.data.customMembers.size() <= 30))
        && (!('msOff' in request.resource.data) || (
              request.resource.data.msOff is list
              && request.resource.data.msOff.size() <= 400));
      allow delete: if canManageDuty();
    }

    /* =====================================================================
       team_events — 행사일(당직 없음). 문서 id = 'YYYY-MM-DD'
       · 배정 전에 등록하면 자동 배정에서 건너뜁니다.
       · 배정 후에 등록하면 그날 당직만 취소됩니다(앱에서 자동 처리).
       ===================================================================== */
    match /team_events/{dateId} {
      allow read: if isApproved();
      allow create, update: if canManageDuty()
        && request.resource.data.date == dateId
        && request.resource.data.date is string
        && request.resource.data.date.size() == 10
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.title.size() <= 30;
      allow delete: if canManageDuty();
    }

    /* =====================================================================
       photos — 사진공유 (이미지는 압축 Data URL 로 문서에 저장)
       ===================================================================== */
    match /photos/{id} {
      allow read: if isApproved();

      allow create: if isApproved()
        && request.resource.data.author_id == request.auth.uid
        && request.resource.data.imageData is string
        && request.resource.data.imageData.size() < 900000   /* 약 1MB 문서 한도 내 */
        && request.resource.data.caption is string
        && request.resource.data.caption.size() <= 200;

      /* 좋아요·댓글은 팀원 누구나, 그 외 수정은 작성자·관리자만 */
      allow update: if isApproved()
        && (
             (resource.data.author_id == request.auth.uid || isAdmin())
             || onlyChanged(['likes','comments'])
           );

      allow delete: if isApproved()
        && (resource.data.author_id == request.auth.uid || isAdmin());
    }

    /* =====================================================================
       chat_messages — 팀 단체 대화
       ===================================================================== */
    match /chat_messages/{id} {
      allow read: if isApproved();

      allow create: if isApproved()
        && request.resource.data.author_id == request.auth.uid
        && request.resource.data.body is string
        && request.resource.data.body.size() > 0
        && request.resource.data.body.size() <= 1000;

      allow update: if false;
      allow delete: if isApproved()
        && (resource.data.author_id == request.auth.uid || isAdmin());
    }

    /* 그 외 경로는 모두 차단 */
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

/* ============================================================================
   참고
   - 조회는 대부분 단순 정렬(orderBy createdAt)이라 복합 인덱스가 필요 없습니다.
   - 사진은 Storage 버킷을 쓰지 않으므로 Blaze(카드 등록) 없이 Spark 로 동작합니다.
     (브라우저에서 가로 960px / JPEG 0.72 로 압축 → 100~250KB 수준)
   - 팀 규모가 작아 Spark 무료 할당량(일일 읽기 5만·쓰기 2만)으로 충분합니다.
   - 앱의 상단 FIREBASE_CONFIG 또는 [설정 → Firebase 연결] 에
     apiKey / authDomain / projectId / storageBucket / messagingSenderId / appId 를
     넣으면 즉시 라이브 모드로 전환됩니다.
   ============================================================================ */
